parameter DATA_WIDTH = 8;
parameter ADDR_WIDTH = 8;

reg [DATA_WIDTH-1:0] bram [0:(1<<ADDR_WIDTH)-1];
[opcode - 4 bit][reg1Addr - 3 bit][reg2Addr - 3 bit]

    DATA_WIDTH = 8 → Each memory word is 8 bits wide
    ADDR_WIDTH = 8 → Address is 8 bits, allowing 2⁸ = 256 locations
    [0:(1<<ADDR_WIDTH)-1] → Memory indices from 0 to 255

    if (we) begin
        bram[addr] <= din;    // write operation
    end
    dout <= bram[addr];       // synchronous read
end


    we = write enable
    addr = address to read/write
    din = data input
    dout = data output

# Fetch-Decode Pipeline
fetch block will fetch the instruction bits of the next instruction
fetch block will have to store the instruction bits in a register by the end of the current cycle

# Decode-Execute Pipeline
decoder outputs 
- the source and destination registers of the instruction
- the immediate value in the instruction
- various control signals (denote the type of the instruction)

append the data from the fetch block, and pipeline all these signals to the execute block.

# Execute-Memory Pipeline
execute block consists of just one ALU
for memory instructions and branch instructions
In memory instructions, the ALU computes the address to read/write data from
In branch instructions, the ALU computes the target address, and the CPU jumps to that target and the CPU jumps
